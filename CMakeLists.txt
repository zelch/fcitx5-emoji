cmake_minimum_required(VERSION 3.16)
project(fcitx5-emoji VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EMOJI_USER_INSTALL "Install in the user's home with an absolute addon library path" OFF)
if(EMOJI_USER_INSTALL)
  if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "$ENV{HOME}/.local" CACHE PATH "Install prefix" FORCE)
  endif()
  set(FCITX_INSTALL_USE_FCITX_SYS_PATHS OFF CACHE BOOL "Install addon files to Fcitx install dir" FORCE)
endif()
include(GNUInstallDirs)
find_package(Fcitx5Core 5.1.3 REQUIRED)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES Fcitx5::Core)
check_cxx_source_compiles("#include <fcitx/candidatelist.h>
class Word : public fcitx::CandidateWord {
public:
  Word() { setComment(fcitx::Text()); }
  void select(fcitx::InputContext *) const override {}
};
int main() { Word word; }" EMOJI_HAS_CANDIDATE_COMMENT)
unset(CMAKE_REQUIRED_LIBRARIES)
find_package(Lua 5.3 REQUIRED)
find_package(ZLIB 1.2.9 REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_library(emoji-engine STATIC src/engine.cpp)
set_target_properties(emoji-engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(emoji-engine PUBLIC src PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(emoji-engine PUBLIC Fcitx5::Core PRIVATE ${LUA_LIBRARIES} ZLIB::ZLIB ICU::uc ICU::i18n)
target_compile_options(emoji-engine PRIVATE -Wall -Wextra -Wpedantic)
add_library(emoji-picker MODULE src/picker.cpp)
target_link_libraries(emoji-picker PRIVATE emoji-engine)
target_compile_definitions(emoji-picker PRIVATE EMOJI_HAS_CANDIDATE_COMMENT=$<BOOL:${EMOJI_HAS_CANDIDATE_COMMENT}>)
target_compile_options(emoji-picker PRIVATE -Wall -Wextra -Wpedantic)
if(EMOJI_USER_INSTALL)
  set(EMOJI_ADDONDIR "${CMAKE_INSTALL_LIBDIR}/fcitx5")
  set(EMOJI_DATADIR "${CMAKE_INSTALL_DATADIR}/fcitx5")
  set(EMOJI_LIBRARY_NAME "${CMAKE_SHARED_MODULE_PREFIX}emoji-picker")
  configure_file(cmake/install-user-descriptor.cmake.in install-user-descriptor.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install-user-descriptor.cmake")
else()
  set(EMOJI_ADDONDIR "${FCITX_INSTALL_ADDONDIR}")
  set(EMOJI_DATADIR "${FCITX_INSTALL_PKGDATADIR}")
  install(FILES data/emoji-picker.conf DESTINATION ${EMOJI_DATADIR}/addon)
endif()
install(TARGETS emoji-picker DESTINATION ${EMOJI_ADDONDIR})
install(FILES lua/search.lua data/metadata.lua DESTINATION ${EMOJI_DATADIR}/emoji-picker)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test-engine tests/engine.cpp)
  target_link_libraries(test-engine PRIVATE emoji-engine ZLIB::ZLIB)
  set(EMOJI_TEST_DICTIONARY_DIR "${FCITX_SYS_INSTALL_PREFIX}/share/fcitx5/emoji/data" CACHE PATH "Installed emoji dictionaries for tests")
  add_test(NAME engine COMMAND test-engine ${CMAKE_CURRENT_SOURCE_DIR} ${EMOJI_TEST_DICTIONARY_DIR})
  add_test(NAME metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_metadata.py --check)
  add_test(NAME metadata-families COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/metadata.py)
  find_package(Fcitx5ModuleTestFrontend REQUIRED)
  find_package(Fcitx5ModuleTestIM REQUIRED)
  add_executable(test-picker tests/picker.cpp)
  target_include_directories(test-picker PRIVATE src)
  target_link_libraries(test-picker PRIVATE Fcitx5::Core Fcitx5::Module::TestFrontend Fcitx5::Module::TestIM)
  add_dependencies(test-picker emoji-picker)
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-data/addon ${CMAKE_CURRENT_BINARY_DIR}/test-data/emoji-picker)
  configure_file(data/emoji-picker.conf test-data/addon/emoji-picker.conf COPYONLY)
  configure_file(lua/search.lua test-data/emoji-picker/search.lua COPYONLY)
  configure_file(data/metadata.lua test-data/emoji-picker/metadata.lua COPYONLY)
  # Include the installed Fcitx dictionaries in this private data tree.
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-data/emoji)
  file(CREATE_LINK ${EMOJI_TEST_DICTIONARY_DIR} ${CMAKE_CURRENT_BINARY_DIR}/test-data/emoji/data SYMBOLIC)
  add_test(NAME picker COMMAND test-picker ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_BINARY_DIR}/test-data)
endif()
install(FILES LICENSE.md data/LICENSE-UNICODE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fcitx5-emoji)
